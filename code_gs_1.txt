var ID_FOLHA = "1zGnYN2fSKZWvtjwho-n5OKWyOvaV1jJ_S7e3CTaonds";
var NOME_ABA_FALTAS = "Faltas";
var NOME_ABA_EQUIPA = "Equipa";

// Aba "Equipa":  A Colaborador | B Password | C Setor | D Dias disponíveis após fecho Somengil
var COL_EQUIPA_NOME = 1;
var COL_EQUIPA_DIAS = 4;

// Aba "Faltas":  A Quem? | B Setor | C Início | D Fim | E Status | F Motivo |
//                G Nº de Dias Pedidos | H Dias Disponíveis |
//                I Tipo | J Horas | K Observações | L Registado em | M Descontado
var COL_FALTAS_DISP = 8;
var COL_FALTAS_DESCONTADO = 13;

// Motivos que consomem dias de férias. TEM DE COINCIDIR com a lista
// MOTIVOS_AUSENCIA (desconta: true) do HTML do portal.
var MOTIVOS_DESCONTAM = ['ferias', 'pessoal'];

// Mensagens: separador "Mensagens" desta mesma folha (A:E  ID | De | Para | Mensagem | Status)
var NOME_ABA_MSG = "Mensagens";


function doGet(e) {
  if (e && e.parameter && e.parameter.teste) {
    return ContentService.createTextOutput(JSON.stringify(escreverAjuste(e.parameter)));
  }
  return ContentService.createTextOutput("OK - endpoint ativo");
}


function doPost(e) {
  var dados = {}, resultado;
  try {
    dados = JSON.parse(e.postData.contents);
    resultado = encaminhar(dados);
  } catch (err) {
    // Fica visível em Execuções > "Registos na nuvem"
    console.error('doPost falhou | acao=' + (dados && dados.acao) + ' | ' + (err && err.stack ? err.stack : err));
    resultado = { ok: false, erro: String(err && err.message ? err.message : err) };
  }
  return ContentService.createTextOutput(JSON.stringify(resultado))
                       .setMimeType(ContentService.MimeType.JSON);
}


function encaminhar(dados) {
  var resultado;
  if (dados.acao && String(dados.acao).indexOf('falta_') === 0) {
    resultado = tratarFaltas(dados);              // pedidos de ausência
  } else if (dados.acao && String(dados.acao).indexOf('msg_') === 0) {
    resultado = tratarMensagens(dados);           // mensagens (separador "Mensagens")
  } else if (dados.acao === 'picagem_nova') {
    resultado = adicionarPicagem(dados);          // nova picagem (botão "+" em ⏰ Picagens)
  } else if (dados.acao === 'eliminar') {
    resultado = eliminarLinha(dados);             // eliminar picagem (como sempre)
  } else {
    resultado = escreverAjuste(dados);            // ajuste de horas (como sempre)
  }
  if (resultado && resultado.ok === false) console.warn('acao=' + dados.acao + ' | ' + JSON.stringify(resultado));
  return resultado;
}

function escreverAjuste(dados) {
  var ss = SpreadsheetApp.openById("1zGnYN2fSKZWvtjwho-n5OKWyOvaV1jJ_S7e3CTaonds");
  var sh = ss.getSheetByName("Dados") || ss.getSheets()[0];
  var cab = sh.getDataRange().getValues()[0];

  function col(nome) {                       // procura tolerante (trim + minúsculas)
    var alvo = String(nome).trim().toLowerCase();
    for (var i = 0; i < cab.length; i++)
      if (String(cab[i]).trim().toLowerCase() === alvo) return i;
    return -1;
  }
  var iColab   = col("Colaborador");
  var iDia     = col("Dia Picagem (hoje)");
  var iHorario = col("Horário de Trabalho");          // ← coluna E
  var iAdd     = col("Ajuste Horas (adicionar)");
  var iRem     = col("Ajuste Horas (retirar)");
  var iSaldo   = col("SALDO");

  var min = Number(dados.minutos) || 0, abs = Math.abs(min);
  var val = ("0" + Math.floor(abs / 60)).slice(-2) + ":" + ("0" + (abs % 60)).slice(-2) + ":00";

  var linha = new Array(cab.length).fill("");
  if (iColab   >= 0) linha[iColab]   = dados.colaborador;
  if (iDia     >= 0) linha[iDia]     = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy");
  if (iHorario >= 0) linha[iHorario] = "Ajuste";      // ← preenche "Ajuste"
  if (min >= 0 && iAdd >= 0) linha[iAdd] = val;
  if (min <  0 && iRem >= 0) linha[iRem] = val;
  if (iSaldo   >= 0) linha[iSaldo]   = (min < 0 ? "-" : "") + val;

  sh.appendRow(linha);
  return { ok: true, colunasEncontradas: { iColab: iColab, iHorario: iHorario, iAdd: iAdd, iRem: iRem, iSaldo: iSaldo } };
}


function eliminarLinha(dados) {
  var lock = LockService.getScriptLock();
  lock.waitLock(10000);
  try {
    var ss = SpreadsheetApp.openById("1zGnYN2fSKZWvtjwho-n5OKWyOvaV1jJ_S7e3CTaonds");
    var sh = ss.getSheetByName("Dados") || ss.getSheets()[0];
    var numLinha = Number(dados.linha);

    if (!numLinha || numLinha < 2 || numLinha > sh.getLastRow()) {
      return { ok: false, erro: "Linha inválida: " + numLinha };
    }

    var cab = sh.getDataRange().getValues()[0];
    function col(nome) {
      var alvo = String(nome).trim().toLowerCase();
      for (var i = 0; i < cab.length; i++)
        if (String(cab[i]).trim().toLowerCase() === alvo) return i;
      return -1;
    }
    var iColab = col("Colaborador");

    // Proteção: confirma que a linha ainda corresponde ao colaborador esperado antes de apagar
    // (evita apagar a linha errada se a folha tiver mudado entre a leitura e o clique)
    if (dados.colaborador && iColab >= 0) {
      var valores = sh.getRange(numLinha, 1, 1, cab.length).getValues()[0];
      var colabFolha = String(valores[iColab] || "").trim().toLowerCase();
      var colabEsperado = String(dados.colaborador).trim().toLowerCase();
      if (colabFolha !== colabEsperado) {
        return { ok: false, erro: "A linha " + numLinha + " já não corresponde a " + dados.colaborador + " — nada foi eliminado." };
      }
    }

    sh.deleteRow(numLinha);
    return { ok: true, linhaEliminada: numLinha };
  } finally {
    lock.releaseLock();
  }
}


/* ============================================================
 *  NOVO — NOVA PICAGEM (aba "Dados", colunas A:AA)
 *
 *  Escreve os campos do formulário (A:R). Para as colunas calculadas
 *  (S:AA): se as linhas anteriores tiverem fórmulas, a fórmula é copiada
 *  para a nova linha; se tiverem valores, escreve os valores calculados
 *  no portal (mesmo formato das linhas existentes).
 *  "Ajuste Horas (adicionar/retirar)" ficam vazias.
 * ============================================================ */

var PIC_HORARIOS = ["Horário Normal (semana)", "Horário Normal + Horas Extra (semana)",
                    "Horas Extra (fim de semana)", "Instalação Técnica"];

function adicionarPicagem(dados) {
  if (!dados.colaborador || !dados.dia) return { ok: false, erro: "Faltam dados (Colaborador ou Dia)." };
  if (PIC_HORARIOS.indexOf(dados.horario) < 0) return { ok: false, erro: "Horário inválido: " + dados.horario };

  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {
    var sh = SpreadsheetApp.openById(ID_FOLHA).getSheetByName("Dados");
    if (!sh) return { ok: false, erro: 'Não existe o separador "Dados".' };
    var largura = sh.getLastColumn();
    var cab = sh.getRange(1, 1, 1, largura).getValues()[0];
    function col(nome) {
      var alvo = normalizar(nome);
      for (var i = 0; i < cab.length; i++) if (normalizar(cab[i]) === alvo) return i;
      return -1;
    }
    // O apóstrofo guarda como TEXTO ("01", não 1): as fórmulas da folha juntam G&":"&H&":"&I
    function dd(v) { var n = Math.max(0, Math.min(59, Number(v) || 0)); return "'" + ("0" + n).slice(-2); }
    var t = dados.totais || {};

    // Campos do formulário
    var entrada = {};
    entrada[col("id")]                         = "'" + (/^[0-9a-f]{8}$/i.test(String(dados.id || "")) ? String(dados.id) : Utilities.getUuid().slice(0, 8));
    entrada[col("Colaborador")]                = String(dados.colaborador);
    entrada[col("Dia Picagem (hoje)")]         = String(dados.dia);
    entrada[col("Horas Trabalho Standard")]    = "08:15:00";
    entrada[col("Horário de Trabalho")]        = dados.horario;
    entrada[col("Horas")]                      = dd(dados.horas);
    entrada[col("Minutos")]                    = dd(dados.minutos);
    entrada[col("Segundos")]                   = "'00";
    entrada[col("Horas Falta")]                = dd(dados.horasFalta);
    entrada[col("Minutos Falta")]              = dd(dados.minutosFalta);
    entrada[col("Segundo Falta")]              = "'00";
    entrada[col("Horas Extra")]                = dd(dados.horasExtra);
    entrada[col("Minutos Extra")]              = dd(dados.minutosExtra);
    entrada[col("Segundos Extra")]             = "'00";

    // Colunas calculadas (só usadas se a folha não tiver fórmulas nessas colunas)
    var calculado = {};
    calculado[col("Banco de Horas")]     = t.banco || "00:00:00";
    calculado[col("Tempo em Falta")]     = t.falta || "00:00:00";
    calculado[col("Tempo Extra (fds)")]  = t.extraFds || "00:00:00";
    calculado[col("Horário Incompleto")] = t.incompleto || "--";
    calculado[col("Tempo Trabalhado")]   = t.trabalhado || "";
    calculado[col("Saldo Horas")]        = t.saldoHoras || "0:00:00";
    calculado[col("SALDO")]              = t.saldo || "00:00:00";

    // Última linha com Colaborador preenchido (getLastRow() conta linhas "vazias" com
    // fórmulas/formatação, o que deixaria buracos e trocaria os nºs de linha no portal)
    var ultima = 1;
    var iCol = col("Colaborador");
    if (sh.getLastRow() >= 2 && iCol >= 0) {
      var nomes = sh.getRange(1, iCol + 1, sh.getLastRow(), 1).getValues();
      for (var k = nomes.length - 1; k >= 1; k--) if (String(nomes[k][0]).trim() !== "") { ultima = k + 1; break; }
    }

    // Procura, nas últimas linhas, a fórmula (R1C1) mais recente de cada coluna
    var formulas = {};
    var linhaModelo = ultima >= 2 ? ultima : 0;
    if (ultima >= 2) {
      var ini = Math.max(2, ultima - 60);
      var bloco = sh.getRange(ini, 1, ultima - ini + 1, largura).getFormulasR1C1();
      for (var c = 0; c < largura; c++) {
        for (var r = bloco.length - 1; r >= 0; r--) {
          if (bloco[r][c]) { formulas[c] = bloco[r][c]; break; }
        }
      }
    }

    var nova = ultima + 1;
    // A folha pode não ter linhas vazias no fim: getRange() fora dos limites dá erro
    // (o appendRow cria a linha sozinho; aqui temos de a criar nós)
    if (nova > sh.getMaxRows()) sh.insertRowsAfter(sh.getMaxRows(), nova - sh.getMaxRows());
    // Mantém a formatação das linhas existentes (ex.: "00" como texto)
    if (linhaModelo) sh.getRange(linhaModelo, 1, 1, largura).copyTo(sh.getRange(nova, 1, 1, largura), SpreadsheetApp.CopyPasteType.PASTE_FORMAT, false);

    var valores = [], comFormula = [];
    for (var j = 0; j < largura; j++) {
      if (entrada.hasOwnProperty(j)) valores.push(entrada[j]);   // campos do formulário mandam sempre
      else if (formulas[j]) { valores.push(""); comFormula.push(j); }
      else if (calculado.hasOwnProperty(j) && j >= 0) valores.push(calculado[j]);
      else valores.push("");
    }
    sh.getRange(nova, 1, 1, largura).setValues([valores]);
    comFormula.forEach(function (j) { sh.getRange(nova, j + 1).setFormulaR1C1(formulas[j]); });

    return { ok: true, acao: dados.acao, linha: nova, id: String(valores[col("id")]).replace(/^'/, ""), colunasComFormula: comFormula.length };
  } finally {
    lock.releaseLock();
  }
}

/* ============================================================
 *  NOVO — MÓDULO DE PEDIDOS DE AUSÊNCIA
 *
 *  Regra do saldo: a coluna D da aba "Equipa" é a fonte de verdade.
 *  - pedido criado (Pendente)  -> não mexe na coluna D
 *  - pedido posto em Aprovado  -> retira os dias da coluna D
 *  - Aprovado -> Pendente/Não Aprovado, ou pedido eliminado -> devolve os dias
 *  A coluna M ("Descontado") marca as linhas já descontadas, para que o
 *  mesmo pedido nunca seja descontado duas vezes.
 *  Só pedidos em DIAS mexem na coluna D; os de horas ficam só registados.
 * ============================================================ */

function tratarFaltas(dados) {
  var ss = SpreadsheetApp.openById(ID_FOLHA);
  var sh = ss.getSheetByName(NOME_ABA_FALTAS);
  if (!sh) return { ok: false, erro: 'Não existe o separador "' + NOME_ABA_FALTAS + '".' };

  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {

    // ---------- novo pedido ----------
    if (dados.acao === 'falta_nova') {
      var saldoAtual = lerDiasEquipa(dados.quem);
      sh.appendRow([
        dados.quem,
        dados.setor,
        dados.inicio,
        dados.fim,
        dados.status || 'Pendente',
        dados.motivo,
        Number(dados.dias) || 0,
        saldoAtual === null ? (Number(dados.disponiveis) || 0) : saldoAtual,
        dados.tipo === 'horas' ? 'Horas' : 'Dias',
        Number(dados.horas) || 0,
        dados.obs || '',
        Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "dd/MM/yyyy HH:mm"),
        ''
      ]);
      var nova = sh.getLastRow();
      var r = sincronizarDesconto(sh, nova);   // por norma nasce Pendente: não desconta
      return { ok: true, acao: dados.acao, linha: nova, saldo: r };
    }

    var num = Number(dados.linha);
    if (!num || num < 2 || num > sh.getLastRow()) {
      return { ok: false, erro: "Linha inválida: " + num };
    }

    // Proteção: a linha tem de continuar a ser do colaborador esperado
    // (os números de linha mudam quando se apagam pedidos)
    if (dados.quem) {
      var quemFolha = normalizar(sh.getRange(num, 1).getValue());
      if (quemFolha !== normalizar(dados.quem)) {
        return { ok: false, erro: "A linha " + num + " já não corresponde a " + dados.quem + " — nada foi alterado." };
      }
    }

    // ---------- editar Status ou Motivo ----------
    if (dados.acao === 'falta_atualizar') {
      var coluna = (dados.campo === 'Motivo') ? 6 : 5;          // F = Motivo | E = Status
      sh.getRange(num, coluna).setValue(dados.valor);
      var res = sincronizarDesconto(sh, num);                   // desconta ou devolve, se for o caso
      return { ok: true, acao: dados.acao, linha: num, campo: dados.campo, valor: dados.valor, saldo: res };
    }

    // ---------- eliminar / anular pedido ----------
    if (dados.acao === 'falta_eliminar') {
      var devolvido = devolverSeDescontado(sh, num);            // se estava aprovado, devolve os dias
      sh.deleteRow(num);
      return { ok: true, acao: dados.acao, linhaEliminada: num, devolvido: devolvido };
    }

    return { ok: false, erro: "Ação desconhecida: " + dados.acao };

  } finally {
    lock.releaseLock();
  }
}


/** Põe a coluna D da "Equipa" de acordo com o estado atual do pedido da linha `num`. */
function sincronizarDesconto(sh, num) {
  var largura = Math.min(COL_FALTAS_DESCONTADO, sh.getMaxColumns());
  var l = sh.getRange(num, 1, 1, largura).getValues()[0];

  var quem   = l[0];
  var status = l[4];
  var motivo = l[5];
  var dias   = Number(l[6]) || 0;
  var tipo   = l[8];
  var marca  = largura >= COL_FALTAS_DESCONTADO ? l[COL_FALTAS_DESCONTADO - 1] : '';

  var jaDescontado = normalizar(marca) === 'sim';
  var deveDescontar = normalizar(status) === 'aprovado'
                   && normalizar(tipo) !== 'horas'
                   && descontaFerias(motivo)
                   && dias > 0;

  if (deveDescontar === jaDescontado) return { semAlteracao: true };

  var delta = deveDescontar ? -dias : dias;
  var novoSaldo = ajustarDiasEquipa(quem, delta);

  if (novoSaldo === null) {
    return { ok: false, erro: 'Colaborador "' + quem + '" não encontrado na aba "' + NOME_ABA_EQUIPA + '".' };
  }

  sh.getRange(num, COL_FALTAS_DESCONTADO).setValue(deveDescontar ? 'Sim' : '');
  sh.getRange(num, COL_FALTAS_DISP).setValue(novoSaldo);
  return { descontado: deveDescontar, dias: dias, novoSaldo: novoSaldo };
}


/** Se a linha já tinha sido descontada, devolve os dias à coluna D (usar antes de apagar). */
function devolverSeDescontado(sh, num) {
  var largura = Math.min(COL_FALTAS_DESCONTADO, sh.getMaxColumns());
  if (largura < COL_FALTAS_DESCONTADO) return 0;
  var l = sh.getRange(num, 1, 1, largura).getValues()[0];
  if (normalizar(l[COL_FALTAS_DESCONTADO - 1]) !== 'sim') return 0;
  var dias = Number(l[6]) || 0;
  if (!dias) return 0;
  ajustarDiasEquipa(l[0], dias);
  return dias;
}


/** Soma `delta` à coluna D da "Equipa" do colaborador. Devolve o novo valor, ou null se não o encontrar. */
function ajustarDiasEquipa(nome, delta) {
  var shEq = SpreadsheetApp.openById(ID_FOLHA).getSheetByName(NOME_ABA_EQUIPA);
  if (!shEq) return null;
  var linha = procurarLinhaEquipa(shEq, nome);
  if (linha < 1) return null;
  var cel = shEq.getRange(linha, COL_EQUIPA_DIAS);
  var novo = Math.round(((Number(cel.getValue()) || 0) + delta) * 100) / 100;
  cel.setValue(novo);
  return novo;
}


/** Lê a coluna D da "Equipa" sem alterar nada. */
function lerDiasEquipa(nome) {
  var shEq = SpreadsheetApp.openById(ID_FOLHA).getSheetByName(NOME_ABA_EQUIPA);
  if (!shEq) return null;
  var linha = procurarLinhaEquipa(shEq, nome);
  if (linha < 1) return null;
  return Number(shEq.getRange(linha, COL_EQUIPA_DIAS).getValue()) || 0;
}


function procurarLinhaEquipa(shEq, nome) {
  var ultima = shEq.getLastRow();
  if (ultima < 1) return -1;
  var nomes = shEq.getRange(1, COL_EQUIPA_NOME, ultima, 1).getValues();
  var alvo = normalizar(nome);
  for (var i = 0; i < nomes.length; i++) {
    if (normalizar(nomes[i][0]) === alvo) return i + 1;
  }
  return -1;
}


/** minúsculas, sem acentos, sem espaços a mais — para comparar nomes e motivos. */
function normalizar(s) {
  var t = String(s === null || s === undefined ? '' : s).trim().toLowerCase().replace(/\s+/g, ' ');
  try { t = t.normalize('NFD').replace(/[\u0300-\u036f]/g, ''); } catch (e) {}
  return t;
}


function descontaFerias(motivo) {
  return MOTIVOS_DESCONTAM.indexOf(normalizar(motivo)) >= 0;
}

  Logger.log('Saldo depois de aprovar: ' + lerDiasEquipa(NOME));

  Logger.log('Eliminar: ' + JSON.stringify(
    tratarFaltas({ acao: 'falta_eliminar', linha: criado.linha, quem: NOME })));
  Logger.log('Saldo final (deve ser igual ao inicial): ' + lerDiasEquipa(NOME));
}


/* ============================================================
 *  NOVO — MÓDULO DE MENSAGENS (separador "Mensagens")
 *  Cabeçalho A:E  ID | De | Para | Mensagem | Status
 *  Ações:  msg_nova     -> acrescenta uma linha (Status "Não lida")
 *          msg_lida     -> põe "Lida" nas mensagens indicadas (só as dirigidas a quem lê)
 *          msg_eliminar -> apaga a linha com esse ID (só o RH)
 * ============================================================ */

function tratarMensagens(dados) {
  var sh = SpreadsheetApp.openById(ID_FOLHA).getSheetByName(NOME_ABA_MSG);
  if (!sh) return { ok: false, erro: 'Não existe o separador "' + NOME_ABA_MSG + '".' };

  // Texto começado por = + - @ seria interpretado como fórmula: guarda-se como texto
  function texto(v) {
    var t = String(v === null || v === undefined ? "" : v);
    return /^[=+\-@]/.test(t) ? "'" + t : t;
  }
  function norm(v) { return String(v || "").trim().toLowerCase(); }

  var lock = LockService.getScriptLock();
  lock.waitLock(20000);
  try {

    // ---------- nova mensagem ----------
    if (dados.acao === 'msg_nova') {
      var msg = String(dados.mensagem || "").trim();
      if (!msg || !dados.de || !dados.para) return { ok: false, erro: "Faltam dados (De, Para ou Mensagem)." };
      if (msg.length > 2000) msg = msg.slice(0, 2000);
      // Os colaboradores só podem escrever ao RH
      if (norm(dados.de) !== 'rh' && norm(dados.para) !== 'rh') return { ok: false, erro: "Os colaboradores só podem escrever ao RH." };
      var id = /^MSG-[0-9A-Za-z-]{1,40}$/.test(String(dados.id || ""))
        ? String(dados.id)
        : "MSG-" + Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyyMMdd-HHmmss") + "-" + Math.random().toString(36).slice(2, 6);
      sh.appendRow([id, texto(dados.de), texto(dados.para), texto(msg), "Não lida"]);
      return { ok: true, acao: dados.acao, id: id };
    }

    var ultima = sh.getLastRow();
    if (ultima < 2) return { ok: true, acao: dados.acao, alteradas: 0 };
    var valores = sh.getRange(2, 1, ultima - 1, 5).getValues();   // A:E sem o cabeçalho

    // ---------- marcar como lidas ----------
    if (dados.acao === 'msg_lida') {
      var ids = {};
      (dados.ids || []).forEach(function (x) { ids[String(x)] = true; });
      var n = 0;
      for (var i = 0; i < valores.length; i++) {
        if (ids[String(valores[i][0])] && norm(valores[i][2]) === norm(dados.leitor) && norm(valores[i][4]) !== 'lida') {
          sh.getRange(i + 2, 5).setValue("Lida");
          n++;
        }
      }
      return { ok: true, acao: dados.acao, alteradas: n };
    }

    // ---------- apagar mensagem (só RH) ----------
    if (dados.acao === 'msg_eliminar') {
      if (norm(dados.por) !== 'rh') return { ok: false, erro: "Só o RH pode apagar mensagens." };
      for (var j = valores.length - 1; j >= 0; j--) {
        if (String(valores[j][0]).trim() === String(dados.id).trim()) {
          sh.deleteRow(j + 2);
          return { ok: true, acao: dados.acao, id: dados.id };
        }
      }
      return { ok: false, erro: "Mensagem não encontrada: " + dados.id };
    }

    return { ok: false, erro: "Ação desconhecida: " + dados.acao };

  } finally {
    lock.releaseLock();
  }
}


/**
 * Teste rápido (correr no editor com ▶): cria uma mensagem de teste no
 * separador "Mensagens" para confirmar permissões de escrita. Depois apaga-a à mão.
 */
function testarMensagens() {
  Logger.log(JSON.stringify(tratarMensagens({
    acao: 'msg_nova', de: 'RH', para: 'TESTE — apagar', mensagem: 'Mensagem de teste'
  })));
}
